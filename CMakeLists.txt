cmake_minimum_required(VERSION 4.4)
project(ipv4fromnoise VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Choose the type of build." FORCE)
endif()
if(MSVC)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU")
    # winget Clang using GCC CLI driver (clang++.exe)
    add_compile_options($<$<CONFIG:Debug>:-g3> $<$<CONFIG:Debug>:-gcodeview>)
    add_link_options($<$<CONFIG:Debug>:-Wl,-debug>)
  else()
    # Native MSVC (cl.exe) OR winget Clang using MSVC CLI driver (clang-cl.exe)
    add_compile_options($<$<CONFIG:Debug>:/Zi>)
    add_link_options($<$<CONFIG:Debug>:/DEBUG>)
  endif()
elseif(UNIX)
  # Linux/Unix standard GCC/Clang
  add_compile_options($<$<CONFIG:Debug>:-g3> $<$<CONFIG:Debug>:-ggdb>)
endif()
add_executable(out main.cpp)
